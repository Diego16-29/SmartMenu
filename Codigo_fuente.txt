package com.smartmenu.tv;

import android.app.Activity;
import android.os.Bundle;
import android.view.KeyEvent;
import android.view.View;
import android.view.WindowManager;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.webkit.WebResourceRequest;

/** Contenedor Android TV de la interfaz local, sin acceso a red. */
public class MainActivity extends Activity {
    private WebView web;
    @Override public void onCreate(Bundle state) {
        super.onCreate(state);
        getWindow().addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON);
        immersive();
        web = new WebView(this);
        web.setBackgroundColor(0xff151310);
        web.getSettings().setJavaScriptEnabled(true);
        web.getSettings().setDomStorageEnabled(true);
        web.getSettings().setAllowFileAccess(false);
        web.getSettings().setAllowContentAccess(false);
        web.getSettings().setBlockNetworkLoads(true);
        web.setWebViewClient(new WebViewClient() {
            @Override public boolean shouldOverrideUrlLoading(WebView v, WebResourceRequest r) { return true; }
            @Override public boolean shouldOverrideUrlLoading(WebView v, String url) { return true; }
        });
        setContentView(web);
        web.setFocusable(true);
        web.setFocusableInTouchMode(true);
        web.requestFocus();
        web.loadUrl("file:///android_asset/index.html");
    }
    private void immersive() {
        getWindow().getDecorView().setSystemUiVisibility(View.SYSTEM_UI_FLAG_FULLSCREEN | View.SYSTEM_UI_FLAG_HIDE_NAVIGATION | View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY | View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN | View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION | View.SYSTEM_UI_FLAG_LAYOUT_STABLE);
    }
    @Override public void onWindowFocusChanged(boolean hasFocus) {
        super.onWindowFocusChanged(hasFocus); if (hasFocus) immersive();
    }
    @Override public boolean dispatchKeyEvent(KeyEvent event) {
        String key;
        switch(event.getKeyCode()) {
            case KeyEvent.KEYCODE_DPAD_UP: key="ArrowUp"; break;
            case KeyEvent.KEYCODE_DPAD_DOWN: key="ArrowDown"; break;
            case KeyEvent.KEYCODE_DPAD_LEFT: key="ArrowLeft"; break;
            case KeyEvent.KEYCODE_DPAD_RIGHT: key="ArrowRight"; break;
            case KeyEvent.KEYCODE_DPAD_CENTER: case KeyEvent.KEYCODE_ENTER: case KeyEvent.KEYCODE_NUMPAD_ENTER: key="Enter"; break;
            case KeyEvent.KEYCODE_BACK: key="Escape"; break;
            default: return super.dispatchKeyEvent(event);
        }
        if (web != null && event.getAction() == KeyEvent.ACTION_DOWN && (event.getRepeatCount()==0 || key.startsWith("Arrow"))) {
            web.evaluateJavascript("window.tvKey && window.tvKey('"+key+"')", result -> {
                if (key.equals("Escape") && "false".equals(result)) finish();
            });
        }
        return true;
    }
    @Override public void onBackPressed() {
        if (web != null) web.evaluateJavascript("window.tvKey && window.tvKey('Escape')", result -> {
            if ("false".equals(result)) finish();
        }); else super.onBackPressed();
    }
    @Override protected void onDestroy() { if(web!=null) { web.destroy(); web=null; } super.onDestroy(); }
}

 const dir={ArrowLeft:[-1,0],ArrowRight:[1,0],ArrowUp:[0,-1],ArrowDown:[0,1]}[key];if(!dir)return false;const center=b=>{const r=b.getBoundingClientRect();return [r.x+r.width/2,r.y+r.height/2];};const [x,y]=center(active);let best=null,score=Infinity;for(const b of buttons){if(b===active)continue;const [bx,by]=center(b),dx=bx-x,dy=by-y,forward=dx*dir[0]+dy*dir[1],cross=Math.abs(dx*dir[1]-dy*dir[0]);if(forward>5){const s=forward+cross*3;if(s<score){score=s;best=b;}}}if(best)best.focus();return true;}
window.tvKey=tvKey;document.addEventListener('keydown',e=>{if(['ArrowLeft','ArrowRight','ArrowUp','ArrowDown','Enter','Escape'].includes(e.key)){e.preventDefault();tvKey(e.key);}});

======================================================================
generar-apk.ps1
======================================================================
$ErrorActionPreference = 'Stop'
Set-Location $PSScriptRoot
if (-not $env:JAVA_HOME -and (Test-Path "$env:ProgramFiles\Android\Android Studio\jbr\bin\java.exe")) { $env:JAVA_HOME = "$env:ProgramFiles\Android\Android Studio\jbr" }
if ($env:JAVA_HOME) { $env:Path = "$env:JAVA_HOME\bin;$env:Path" }
if (-not (Get-Command java -ErrorAction SilentlyContinue)) { throw 'Instala Android Studio o JDK 17 y configura JAVA_HOME.' }
$sdk = $env:ANDROID_HOME
if (-not $sdk) { $sdk = $env:ANDROID_SDK_ROOT }
if (-not $sdk) { $sdk = "$env:LOCALAPPDATA\Android\Sdk" }
if (-not (Test-Path "$sdk\platforms\android-35\android.jar")) { throw 'Instala Android SDK Platform 35 desde SDK Manager en Android Studio.' }
$env:ANDROID_HOME = $sdk
$env:ANDROID_SDK_ROOT = $sdk
$gradleRoot = Join-Path $env:LOCALAPPDATA 'SmartMenuTools'
$gradle = Join-Path $gradleRoot 'gradle-8.9\bin\gradle.bat'
if (-not (Test-Path $gradle)) {
  New-Item -ItemType Directory -Force $gradleRoot | Out-Null
  $archive = Join-Path $gradleRoot 'gradle-8.9-bin.zip'
  Invoke-WebRequest -Uri 'https://services.gradle.org/distributions/gradle-8.9-bin.zip' -OutFile $archive
  Expand-Archive -Path $archive -DestinationPath $gradleRoot -Force
}
& $gradle --no-daemon wrapper --gradle-version 8.9
if ($LASTEXITCODE -ne 0) { throw 'No se pudo generar el Gradle Wrapper.' }
& $gradle --no-daemon :app:assembleDebug
if ($LASTEXITCODE -ne 0) { throw 'Error de compilacion de Android.' }
Copy-Item 'app\build\outputs\apk\debug\app-debug.apk' 'SmartMenu-debug.apk' -Force
Write-Host "APK generado: $PSScriptRoot\SmartMenu-debug.apk" -ForegroundColor Green
